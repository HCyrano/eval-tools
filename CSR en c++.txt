✅ Option 1 — Fichiers texte simples (le plus facile)
C++ : écrire les fichiers
#include <vector>
#include <fstream>
#include <iostream>

void save_csr_to_text(
    const std::string& basename,
    const std::vector<double>& data,
    const std::vector<int>& indices,
    const std::vector<int>& indptr,
    int nrows, int ncols)
{
    std::ofstream f_data(basename + "_data.txt");
    for (double v : data) f_data << v << "\n";

    std::ofstream f_indices(basename + "_indices.txt");
    for (int c : indices) f_indices << c << "\n";

    std::ofstream f_indptr(basename + "_indptr.txt");
    for (int p : indptr) f_indptr << p << "\n";

    std::ofstream f_shape(basename + "_shape.txt");
    f_shape << nrows << " " << ncols << "\n";

    std::cout << "CSR files written with basename '" << basename << "'\n";
}

int main() {
    // Exemple : matrice 3x3
    // [10 0  0 ]
    // [ 0 20 30]
    // [40  0 50]
    std::vector<double> data   = {10, 20, 30, 40, 50};
    std::vector<int>    indices= {0, 1, 2, 0, 2};
    std::vector<int>    indptr = {0, 1, 3, 5};
    int nrows = 3, ncols = 3;

    save_csr_to_text("A", data, indices, indptr, nrows, ncols);
    return 0;
}

Python : charger la matrice
import numpy as np
from scipy.sparse import csr_matrix

data = np.loadtxt("A_data.txt")
indices = np.loadtxt("A_indices.txt", dtype=np.int32)
indptr = np.loadtxt("A_indptr.txt", dtype=np.int32)
nrows, ncols = np.loadtxt("A_shape.txt", dtype=int)

A = csr_matrix((data, indices, indptr), shape=(nrows, ncols))
print(A.toarray())

 [ 0. 20. 30.]
 [40.  0. 50.]]

⚡ Option 2 — Format .npz (rapide et natif NumPy/SciPy)

Tu peux aussi écrire les fichiers binaires .npy depuis C++ si tu veux des I/O rapides.
Pour ça, le plus simple est d’utiliser la petite bibliothèque cnpy
 (header-only, open source, très utilisée).

Exemple C++ avec cnpy
#include "cnpy.h"  // https://github.com/rogersce/cnpy
#include <vector>

int main() {
    std::vector<double> data   = {10, 20, 30, 40, 50};
    std::vector<int>    indices= {0, 1, 2, 0, 2};
    std::vector<int>    indptr = {0, 1, 3, 5};
    int nrows = 3, ncols = 3;

    cnpy::npy_save("data.npy", data.data(), {data.size()}, "w");
    cnpy::npy_save("indices.npy", indices.data(), {indices.size()}, "w");
    cnpy::npy_save("indptr.npy", indptr.data(), {indptr.size()}, "w");

    FILE* f = fopen("shape.txt", "w");
    fprintf(f, "%d %d\n", nrows, ncols);
    fclose(f);

    return 0;
}


Puis en Python :

import numpy as np
from scipy.sparse import csr_matrix

data = np.load("data.npy")
indices = np.load("indices.npy")
indptr = np.load("indptr.npy")
nrows, ncols = np.loadtxt("shape.txt", dtype=int)

A = csr_matrix((data, indices, indptr), shape=(nrows, ncols))
print(A.toarray())


🟣 Avantages :

Lecture/écriture binaire très rapide

Aucun parsing texte

Format compatible directement avec numpy.load

 [ 0. 20. 30.]
 [40.  0. 50.]]
